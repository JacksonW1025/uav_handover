cmake_minimum_required(VERSION 3.16)
project(handover_ros_probe)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(px4_msgs REQUIRED)
find_package(px4_ros2_cpp QUIET)
add_executable(ros_probe main.cpp)
ament_target_dependencies(ros_probe rclcpp px4_msgs)
if(px4_ros2_cpp_FOUND)
  ament_target_dependencies(ros_probe px4_ros2_cpp)
  target_compile_definitions(ros_probe PRIVATE HAVE_PX4_INTERFACE)
endif()
